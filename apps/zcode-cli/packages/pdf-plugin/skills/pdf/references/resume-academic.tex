% Academic resume / CV template — original work for this repository.
%
% A one-column academic CV: conservative typography, sections that hold
% education, appointments, publications, grants, teaching and service. Public
% LaTeX knowledge only; no third-party template is vendored. Requires a TeX
% distribution with `moderncv`-free plain article class, `geometry`,
% `titlesec`, `enumitem`, `hyperref`, `fontenc`.
%
% Build:  latexmk -pdf resume-academic.tex
% Page:   US Letter by default; switch via the geometry options below.

\documentclass[11pt]{article}

% ---------------------------------------------------------------- geometry
\usepackage[letterpaper,margin=1in]{geometry}
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{microtype}
\usepackage{titlesec}
\usepackage{enumitem}
\usepackage{fancyhdr}
\usepackage{xcolor}
\usepackage[hidelinks]{hyperref}

% ---------------------------------------------------------------- type
\renewcommand{\familydefault}{\rmdefault}   % serif body: the academic default
\setlength{\parindent}{0pt}
\setlist[itemize]{leftmargin=1.4em, itemsep=1pt, parsep=0pt, topsep=2pt}

% ---------------------------------------------------------------- headings
\titleformat{\section}{\large\bfseries\scshape}{}{0em}{}[\titlerule]
\titlespacing{\section}{0pt}{12pt}{6pt}
\titleformat{\subsection}{\normalsize\bfseries}{}{0em}{}

% Running head: name on the left, page number on the right.
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\small\scshape Curriculum Vitae}
\fancyhead[R]{\small\thepage}
\renewcommand{\headrulewidth}{0.4pt}

% ---------------------------------------------------------------- helpers
% An entry line: bold left item, right-aligned date, no table rules.
\newcommand{\cventry}[3]{%
  \par\noindent\textbf{#1}\hfill{\small #2}\par
  \noindent #3\par\vspace{4pt}%
}
% A dated list item (publication, grant, talk).
\newcommand{\cvitem}[2]{\item \textbf{#1.} #2}

\begin{document}

   \href{https://orcid.org/0000-0000-0000-0000}{ORCID}}
\end{center}
\vspace{2pt}
